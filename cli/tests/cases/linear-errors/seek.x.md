# Goal seek forms

[squared] := solve(squared * squared >= 4)
[inverse] := solve(1 / inverse >= 2)
[bare] := solve(bare + 1)
[lone] := solve(lone)
[called] := solve(round(called))
[texty] := solve(texty + "a" >= 1)
[negated] := solve(!negated >= 1)
[nested] := solve((nested < 1) + 1 >= 2)
[zero] := solve(zero / 0 >= 1)
[mixed] := solve(mixed * $0 + mixed * 2h >= 1h)
[scaled] := solve(scaled * $2 * 2h >= $1)
[divided] := solve(divided / 2h >= 1)
[compared] := solve(compared * $1 >= 2h)
[euros] := solve(euros * $1 + €2 >= $5)
[cancels] := solve(cancels - cancels >= 1)
[budget] := solve(100 - 4 * budget >= 20)
[hours] := solve(hours * $25 <= $190)
[exact] := solve(3 * exact + 2 == 11)
